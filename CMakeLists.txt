cmake_minimum_required(VERSION 3.22)
project(amber_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOTCPP_API_VERSION "4.6" CACHE STRING "" FORCE)
set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/native/profile.json" CACHE FILEPATH "" FORCE)
set(GODOT_CPP_DIR "" CACHE PATH "Official godot-cpp 10.0.0-stable checkout")
set(LUA_SOURCE_DIR "" CACHE PATH "Unmodified upstream Lua 5.4.7 sources")
set(FSNET_SOURCE_DIR "" CACHE PATH "External fsnet headers; not bundled")
set(FSNET_LIBRARY "" CACHE FILEPATH "External built libfsnet")
if(NOT EXISTS "${GODOT_CPP_DIR}/CMakeLists.txt" OR NOT EXISTS "${LUA_SOURCE_DIR}/lua.h" OR NOT EXISTS "${FSNET_SOURCE_DIR}/include/fsnet/node.hpp" OR NOT EXISTS "${FSNET_LIBRARY}")
  message(FATAL_ERROR "Set GODOT_CPP_DIR, LUA_SOURCE_DIR, FSNET_SOURCE_DIR, FSNET_LIBRARY; see README")
endif()
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
file(GLOB LUA_SOURCES "${LUA_SOURCE_DIR}/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac|onelua|ltests)\\.c$")
add_library(amber_lua STATIC ${LUA_SOURCES})
target_include_directories(amber_lua PUBLIC "${LUA_SOURCE_DIR}")
add_library(amber SHARED native/extension.cpp)
target_include_directories(amber PRIVATE "${FSNET_SOURCE_DIR}/include")
target_link_libraries(amber PRIVATE godot::cpp amber_lua "${FSNET_LIBRARY}")
if(WIN32)
  set_target_properties(amber PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/windows/x86_64")
endif()
set_target_properties(amber PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin" OUTPUT_NAME amber)

set_target_properties(amber amber_lua PROPERTIES CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
